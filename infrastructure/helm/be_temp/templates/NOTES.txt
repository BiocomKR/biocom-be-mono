{{- else if contains "NodePort" .Values.service.type }}
  
  NodePort를 통한 접속:
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "be-temp.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
  
{{- else if contains "LoadBalancer" .Values.service.type }}
  
  LoadBalancer를 통한 접속:
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "be-temp.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
  
{{- else if contains "ClusterIP" .Values.service.type }}
  
  포트 포워딩을 통한 로컬 접속:
  kubectl port-forward svc/{{ include "be-temp.fullname" . }} 8080:{{ .Values.service.port }} -n {{ .Release.Namespace }}
  http://localhost:8080
  
{{- end }}
